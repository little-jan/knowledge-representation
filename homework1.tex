\indent $h1 = A \lor B$ \\
\indent $h2 = \neg B$

    \RightLabel{($h1$)}
    \UnaryInfC{$A \lor B$} 
    
    \AxiomC{$[A]^1$}
    
    \AxiomC{$[B]^2$}
    \AxiomC{} 
    \RightLabel{($h2$)} 
    \UnaryInfC{$\neg B$}
    \RightLabel{($\neg$E)}
    \BinaryInfC{$\bot$}
    \RightLabel{($\bot$E)}
    \UnaryInfC{$A$}
    
    \RightLabel{($\lor$E)$^{1,2}$}
    \TrinaryInfC{$A$}
    
    \RightLabel{($\implies$I)$^{h1, h2}$}
    \UnaryInfC{$(A \lor B) \land \neg B \implies A$}
